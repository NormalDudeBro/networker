config router ospf
    set router-id 1.1.1.1
    set default-information-originate enable
    set auto-cost-ref-bandwidth 1000
    config area
        edit 0.0.0.0
        next
        edit 0.0.0.10
        next
    end
    config ospf-interface
        edit "ospf-net-1"
            set interface "port1"
        next
        edit "ospf-net-2"
            set interface "port1"
        next
    end
    config network
        edit 1
            set prefix 10.0.0.0 255.255.255.0
            set area 0.0.0.0
        next
        edit 2
            set prefix 192.168.100.0 255.255.255.0
            set area 0.0.0.10
        next
    end
end

config router bgp
    set as 65000
    set router-id 1.1.1.1
    set log-neighbour-changes enable
    config neighbor
        edit "10.0.0.2"
            set remote-as 65001
            set description "core-peer"
            set password s3cret
            set update-source "Loopback0"
            set ebgp-multihop-ttl 2
            set route-map-in "RM-INTERNAL"
        next
        edit "10.0.1.2"
            set remote-as 65000
            set description "ibgp-peer"
        next
    end
    config network
        edit 1
            set prefix 10.0.0.0/8
        next
        edit 2
            set prefix 192.168.100.0/24
        next
    end
    config redistribute "connected"
        set status enable
    end
    config redistribute "static"
        set status enable
    end
    config redistribute "ospf"
    end
end

config router prefix-list
    edit "PL-INTERNAL"
        config rule
            edit 10
                set action permit
                set prefix 10.0.0.0/8
                set ge 16
                set le 24
            next
            edit 20
                set action deny
                set prefix 0.0.0.0/0
            next
        end
    next
end

config router route-map
    edit "RM-INTERNAL"
        config rule
            edit 10
                set action permit
                set match-ip-address "PL-INTERNAL"
                set set-local-preference 150
                set set-metric 50
                set set-community "65000:100"
            next
        end
    next
end
